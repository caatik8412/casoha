Authorize when prompted. Copy the /exec URL. * 5. Paste that URL into CONFIG.SHEET_URL in the HTML app. * Re-deploy (Manage deployments -> edit -> new version) after any edit. ****************************************************************/ function getSheet(name, headers) { headers = headers || []; const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); if (headers.length) { sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); sh.setFrozenRows(1); } } return sh; } function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return ok({ ok: false, error: 'No POST payload.' }); } return handleWrite(JSON.parse(e.postData.contents)); } catch (err) { return ok({ ok: false, error: String(err) }); } } function handleWrite(data) { try { if (data.type === 'session') { const sh = getSheet('Sessions', ['Logged At', 'Study Date', 'Paper', 'Minutes', 'Hours', 'Note', 'Session ID']); const ids = sh.getLastRow() > 1 ? sh.getRange(2, 7, sh.getLastRow() - 1, 1).getValues().flat().map(String) : []; if (ids.indexOf(String(data.id)) === -1) { sh.appendRow([ new Date(data.ts), data.date, data.paper, data.mins, (data.mins / 60).toFixed(2), data.note || '', String(data.id) ]); } } else if (data.type === 'state') { const sh = getSheet('State', ['Updated', 'JSON']); const cur = readState_(sh); writeState_(sh, { targets: data.targets != null ? data.targets : cur.targets, portion: data.portion != null ? data.portion : cur.portion, dailyGoal: data.dailyGoal != null ? data.dailyGoal : cur.dailyGoal, weeklyHrs: data.weeklyHrs != null ? data.weeklyHrs : cur.weeklyHrs, tasks: mergeById_(cur.tasks, data.tasks), plan: mergeById_(cur.plan, data.plan) }); } else if (data.type === 'guideAdd') { const sh = getSheet('State', ['Updated', 'JSON']); const cur = readState_(sh); if (data.targets) cur.targets = data.targets; if (data.dailyGoal != null) cur.dailyGoal = data.dailyGoal; if (data.weeklyHrs != null) cur.weeklyHrs = data.weeklyHrs; if (data.planItem) cur.plan = mergeById_(cur.plan, [data.planItem]); if (data.taskItem) cur.tasks = mergeById_(cur.tasks, [data.taskItem]); writeState_(sh, cur); } else if (data.type === 'live') { const sh = getSheet('Live', ['Updated', 'Status', 'Paper', 'Started At', 'Elapsed Min', 'Note']); const row = [ new Date(), data.status || 'idle', data.paper || '', data.startedAt ? new Date(data.startedAt) : '', data.elapsed != null ? data.elapsed : '', data.note || '' ]; if (sh.getLastRow() < 2) sh.appendRow(row); else sh.getRange(2, 1, 1, row.length).setValues([row]); } return ok({ ok: true }); } catch (err) { return ok({ ok: false, error: String(err) }); } } // Optional: lets you (or the app) pull state/sessions back if you rebuild a device function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); // WRITE-VIA-GET: the app sends ?data= because no-cors POST gets dropped by the browser. // doGet always fires reliably, so all writes are routed here. if (e && e.parameter && e.parameter.data) { return handleWrite(JSON.parse(e.parameter.data)); } // DIAGNOSTIC: ?test=write -> writes a row to Sessions straight from the browser. if (e && e.parameter && e.parameter.test === 'write') { if (!ss) return ok({ ok: false, error: 'NOT BOUND: getActiveSpreadsheet() is null. The script is standalone, not attached to a Sheet.' }); const sh = getSheet('Sessions', ['Logged At', 'Study Date', 'Paper', 'Minutes', 'Hours', 'Note', 'Session ID']); sh.appendRow([new Date(), new Date().toLocaleDateString('en-CA'), 'TEST', 1, '0.02', 'browser write test', 'test' + Date.now()]); return ok({ ok: true, msg: 'Test row written to Sessions tab of: ' + ss.getName() }); } // monitor page polls this: ?live=1 -> just the live status, lightweight if (e && e.parameter && e.parameter.live) { const lv = ss.getSheetByName('Live'); if (lv && lv.getLastRow() >= 2) { const r = lv.getRange(2, 1, 1, 6).getValues()[0]; return ok({ updated: new Date(r[0]).getTime(), status: r[1], paper: r[2], startedAt: r[3] ? new Date(r[3]).getTime() : null, elapsed: r[4], note: r[5] }); } return ok({ status: 'idle', updated: null }); } const out = {}; const st = ss.getSheetByName('State'); if (st && st.getLastRow() >= 2) out.state = JSON.parse(st.getRange(2, 2).getValue()); const ses = ss.getSheetByName('Sessions'); if (ses && ses.getLastRow() > 1) { const rows = ses.getRange(2, 1, ses.getLastRow() - 1, 7).getValues(); out.sessions = rows.map(r => ({ ts: new Date(r[0]).getTime(), date: r[1], paper: r[2], mins: r[3], note: r[5], id: r[6] })); } return ok(out); } catch (err) { return ok({ ok: false, error: String(err) }); } } function ok(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ---- state helpers ---- */ function readState_(sh) { const base = { targets:{FR:170,AFM:170,AAPE:170}, portion:{FR:0,AFM:0,AAPE:0}, dailyGoal:6, weeklyHrs:35, tasks:[], plan:[] }; if (sh.getLastRow() >= 2) { try { return Object.assign(base, JSON.parse(sh.getRange(2, 2).getValue())); } catch (e) {} } return base; } function writeState_(sh, obj) { const payload = JSON.stringify(obj); if (sh.getLastRow() < 2) sh.appendRow([new Date(), payload]); else sh.getRange(2, 1, 1, 2).setValues([[new Date(), payload]]); } // merge two arrays of {id,...}; incoming wins on conflict, keeps items only in current function mergeById_(current, incoming) { if (!incoming) return current || []; if (!current) return incoming || []; const map = {}; current.forEach(x => { if (x && x.id != null) map[x.id] = x; }); incoming.forEach(x => { if (x && x.id != null) map[x.id] = x; }); return Object.keys(map).map(k => map[k]); }
